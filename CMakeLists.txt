cmake_minimum_required(VERSION 3.20)
project(PP2026_EN_Lab02 LANGUAGES CXX)
find_package(OpenMP REQUIRED COMPONENTS CXX)
add_library(lab02_core src/mandelbrot.cpp src/parallel.cpp)
target_include_directories(lab02_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(lab02_core PUBLIC cxx_std_20)
target_link_libraries(lab02_core PUBLIC OpenMP::OpenMP_CXX)
if(MSVC)
    target_compile_options(lab02_core PRIVATE /W4)
else()
    target_compile_options(lab02_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(lab02 src/main.cpp)
target_link_libraries(lab02 PRIVATE lab02_core)
add_executable(lab02_tests tests/tests.cpp)
target_link_libraries(lab02_tests PRIVATE lab02_core)
enable_testing()
add_test(NAME lab02_validation COMMAND lab02_tests)
add_test(NAME lab02_profile_smoke COMMAND lab02 --mode profile --width 7 --height 5 --iterations 12 --passes 2)
add_test(NAME lab02_invalid_dimensions COMMAND lab02 --mode profile --width 1 --height 7)
set_tests_properties(lab02_invalid_dimensions PROPERTIES WILL_FAIL TRUE)

# Preserve Release optimization while emitting MSVC symbols for Visual Studio CPU Usage.
# The executable and library must both carry debug information.
if(MSVC)
    foreach(target_name IN ITEMS lab02_core lab02 lab02_tests)
        target_compile_options(${target_name} PRIVATE $<$<CONFIG:Release>:/Zi>)
    endforeach()
    foreach(target_name IN ITEMS lab02 lab02_tests)
        target_link_options(${target_name} PRIVATE $<$<CONFIG:Release>:/DEBUG:FULL>)
    endforeach()
endif()
